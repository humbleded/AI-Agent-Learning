def prepare_request(
    request: object,
) -> tuple[dict[str, str] | None, dict[str, object] | None]:
    if not isinstance(request, dict):
        return None, make_result(
            status="invalid_input",
            summary="请求必须是字典对象。",
            sources=[],
        )
    input_type = request.get("input_type")
    value = request.get("value")
    if input_type not in ALLOWED_INPUT_TYPES:
        return None, make_result(
            status="invalid_input",
            summary=f"不支持的输入类型：{input_type}",
            sources=[],
        )
    if not isinstance(value, str) or not value.strip():
        return None, make_result(
            status="invalid_input",
            summary="输入值必须是非空字符串。",
            sources=[],
        )
    if input_type == "topic" and len(value) > MAX_TOPIC_LENGTH:
        return None, make_result(
            status="invalid_input",
            summary=f"主题长度不能超过 {MAX_TOPIC_LENGTH} 个字符。",
            sources=[],
        )
    if input_type == "relative_path":
        resolved_path = (SANDBOX / value).resolve()
        if not resolved_path.is_file() or SANDBOX not in resolved_path.parents:
            return None, make_result(
                status="invalid_input",
                summary="相对路径无效或越界。",
                sources=[],
